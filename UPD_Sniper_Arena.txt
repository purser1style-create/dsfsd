local UndoStack = {}
local MAX_UNDO  = 10
local function pushUndoSnapshot(settingsRef,toggleSyncFns)
    local snapshot={Settings={},toggleStates={}}
    for k,v in pairs(settingsRef) do snapshot.Settings[k]=v end
    for k,entry in pairs(toggleSyncFns or {}) do
        snapshot.toggleStates[k]=(entry.getState and entry.getState()) or false
    end
    table.insert(UndoStack,snapshot)
    if #UndoStack>MAX_UNDO then table.remove(UndoStack,1) end
end
local function popUndoSnapshot()
    if #UndoStack==0 then return nil end
    return table.remove(UndoStack,#UndoStack)
end

local BoneWeights={{name="Head",weight=50},{name="UpperTorso",weight=30},{name="Torso",weight=30},{name="LowerTorso",weight=20},{name="HumanoidRootPart",weight=10}}
local function RaycastVisible(char,partName)

    local res=workspace:Raycast(camPos,part.Position-camPos,rp); return res==nil
end
local function GetHeadTarget(char)
    if not Settings.WallCheck then return char:FindFirstChild("Head") end
    if RaycastVisible(char,"Head") then return char:FindFirstChild("Head") end; return nil
end
local function GetVisibleBonesWeighted(char)
    local lc=LocalPlayer.Character; if not lc or not char then return {} end
    local visible={}
    for _,entry in ipairs(BoneWeights) do
        local part=char:FindFirstChild(entry.name)
        if part then
            if not Settings.WallCheck then table.insert(visible,{part=part,weight=entry.weight})
            elseif RaycastVisible(char,entry.name) then table.insert(visible,{part=part,weight=entry.weight}) end
        end
    end
    return visible
end
local function WeightedRandomBone(visibleBones)
    if #visibleBones==0 then return nil end
    local total=0; for _,b in ipairs(visibleBones) do total=total+b.weight end
    local roll=math.random()*total; local cum=0
    for _,b in ipairs(visibleBones) do cum=cum+b.weight; if roll<=cum then return b.part end end
    return visibleBones[#visibleBones].part
end
local function HasFlagOnHead(char)
    local head=char:FindFirstChild("Head"); if not head then return false end
    for _,acc in ipairs(char:GetChildren()) do
        if acc:IsA("Accessory") or acc:IsA("Hat") then
            local handle=acc:FindFirstChild("Handle"); if handle then local d=(handle.Position-head.Position).Magnitude; if d<4.5 then return true end end
        end
        if acc:IsA("BillboardGui") then return true end
    end
    return false
end
local function GetScreenCenter() return Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2) end
local function IsInFOV(head)
    local center=GetScreenCenter(); local sp,on=Camera:WorldToViewportPoint(head.Position)
    if not on then return false,0 end
    local d=(Vector2.new(sp.X,sp.Y)-center).Magnitude; return d<=Settings.FOVRadius,d
end
local function ResolveTargetPart(char)
    local hasFlag=HasFlagOnHead(char); local fm=Settings.FireMode
    if fm==1 or hasFlag then return GetHeadTarget(char)
    else return WeightedRandomBone(GetVisibleBonesWeighted(char)) end
end
local function GetBestTarget()
    local best,bestD=nil,math.huge
    for _,player in ipairs(Players:GetPlayers()) do
        if player==LocalPlayer then continue end
        local char=player.Character; if not char then continue end
        local hum=char:FindFirstChildOfClass("Humanoid"); if not hum or hum.Health<=0 then continue end
        local head=char:FindFirstChild("Head"); if not head then continue end
        local inFOV,d=IsInFOV(head); if not inFOV or d>=bestD then continue end
        local tp=ResolveTargetPart(char); if not tp then continue end
        best={Player=player,Char=char,TargetPart=tp}; bestD=d
    end
    return best
end
local function GetAllTargetsInFOV()
    local targets={}
    for _,player in ipairs(Players:GetPlayers()) do
        if player==LocalPlayer then continue end
        local char=player.Character; if not char then continue end
        local hum=char:FindFirstChildOfClass("Humanoid"); if not hum or hum.Health<=0 then continue end
        local head=char:FindFirstChild("Head"); if not head then continue end
        local inFOV,d=IsInFOV(head); if not inFOV then continue end
        local tp=ResolveTargetPart(char); if not tp then continue end
        table.insert(targets,{Player=player,Char=char,TargetPart=tp,dist=d})
    end
    return targets
end
local function GetMagnetTarget()
    local lc=LocalPlayer.Character; if not lc then return nil end
    local camCF=Camera.CFrame; local origin=camCF.Position; local dir=camCF.LookVector*2000
    local bestBone,bestDist=nil,Settings.MagnetRadius
    for _,player in ipairs(Players:GetPlayers()) do
        if player==LocalPlayer then continue end
        local char=player.Character; if not char then continue end
        local hum=char:FindFirstChildOfClass("Humanoid"); if not hum or hum.Health<=0 then continue end
        for _,entry in ipairs(BoneWeights) do
            local part=char:FindFirstChild(entry.name); if not part then continue end
            local bonePos=part.Position
            local t=math.clamp((bonePos-origin):Dot(dir)/dir:Dot(dir),0,1)
            local closest=origin+(dir*t); local distToRay=(bonePos-closest).Magnitude
            if distToRay<bestDist then bestDist=distToRay; bestBone=bonePos end
        end
    end
    return bestBone
end

                    end
                    local r1=vv(...)
                    if typeof(r1)=="Vector3" and r1.Magnitude>10 then return r1*Settings.BulletVelMult end
                    return r1
                end)
            end)
        end
    end
    return true
end
if not trySilentAimHook() then
    task.spawn(function() for _=1,12 do task.wait(2); if trySilentAimHook() then break end end end)
end

local lastMeleeTime=0
local function hookMelee()
    local RS=game:GetService("ReplicatedStorage"); local meleeRemote=nil
    for _,desc in ipairs(RS:GetDescendants()) do
        if desc:IsA("RemoteEvent") or desc:IsA("RemoteFunction") then
            local nm=desc.Name:lower()
            if nm:find("melee") or nm:find("knife") or nm:find("slash") or nm:find("stab") then meleeRemote=desc; break end
        end
    end
    if meleeRemote and meleeRemote:IsA("RemoteEvent") then
        local orig=hookfunction(meleeRemote.FireServer,function(self,...)
            local now=tick(); if now-lastMeleeTime<Settings.MeleeSpeed then return end; lastMeleeTime=now
            if Settings.MeleeSnap then local enemy=GetBestTarget(); if enemy then local head=enemy.Char:FindFirstChild("Head"); if head then return orig(self,head.Position,...) end end end
            return orig(self,...)
        end)
        return true
    end
    local function watchTool(tool)
        if not tool then return end; local handle=tool:FindFirstChild("Handle"); if not handle then return end
        local origSize=handle.Size
        if Settings.MeleeWide then pcall(function() handle.Size=Vector3.new(origSize.X*2.5,origSize.Y*2.5,origSize.Z*1.5) end) end
        handle.Touched:Connect(function(hit)
            local now=tick(); if now-lastMeleeTime<Settings.MeleeSpeed then return end; lastMeleeTime=now
            if not Settings.MeleeSnap then return end
            local hitChar=hit:FindFirstAncestorOfClass("Model"); if not hitChar then return end
            local hum=hitChar:FindFirstChildOfClass("Humanoid"); if not hum or hum.Health<=0 then return end
            if hitChar==LocalPlayer.Character then return end
            local head=hitChar:FindFirstChild("Head"); if not head then return end
            local origPos=handle.CFrame; pcall(function() handle.CFrame=CFrame.new(head.Position) end)
            task.wait(0.03); pcall(function() handle.CFrame=origPos end)
        end)
    end
    local char=LocalPlayer.Character
    if char then for _,item in ipairs(char:GetChildren()) do if item:IsA("Tool") then watchTool(item) end end end
    LocalPlayer.CharacterAdded:Connect(function(newChar)
        newChar.ChildAdded:Connect(function(item) if item:IsA("Tool") then task.wait(0.2); watchTool(item) end end)
        for _,item in ipairs(newChar:GetChildren()) do if item:IsA("Tool") then task.wait(0.2); watchTool(item) end end
    end)
    return true
end
task.spawn(function() task.wait(2.5); hookMelee() end)

local lastFireTime=0; local isFiring=false
local function doMouseClick()
    if mouse1press and mouse1release then mouse1press(); task.wait(0.03); mouse1release(); return true end
    if VirtualInput then
        local vp=Camera.ViewportSize
        pcall(function() VirtualInput:SendMouseButtonEvent(vp.X/2,vp.Y/2,0,true,game,1) end)
        task.wait(0.03)
        pcall(function() VirtualInput:SendMouseButtonEvent(vp.X/2,vp.Y/2,0,false,game,1) end)
        return true
    end
    return false
end

RunService.Heartbeat:Connect(function()
    if not Settings.AutoFire then isFiring=false; return end
    local now=tick(); if now-lastFireTime<Settings.AutoFireRate then return end
    if isFiring then return end
    if Settings.MultiTarget then
        local targets=GetAllTargetsInFOV(); if #targets==0 then return end
        isFiring=true
        task.spawn(function()
            for _,t in ipairs(targets) do
                if not Settings.AutoFire then break end
                local head=t.Char and t.Char:FindFirstChild("Head")
                if head then local inFOV,_=IsInFOV(head); if not inFOV then continue end end
                if Settings.AimEnabled or Settings.BulletMagnet then SilentTarget={Position=t.TargetPart.Position} end
                doMouseClick(); SilentTarget=nil; task.wait(Settings.AutoFireRate*0.5)
            end
            isFiring=false
        end)
    else
        local enemy=GetBestTarget(); if not enemy then return end
        if Settings.AimEnabled then SilentTarget={Position=enemy.TargetPart.Position} end
        isFiring=true
        task.spawn(function()
            local head=enemy.Char and enemy.Char:FindFirstChild("Head")
            if head then local inFOV,_=IsInFOV(head); if not inFOV then SilentTarget=nil; isFiring=false; return end end
            doMouseClick(); SilentTarget=nil; isFiring=false
        end)
    end
    lastFireTime=now
end)
